// Collapses duplicate bridge instances.
//
// The loader cannot stop an old instance. stopOld() finds the placeholder stop
// handle that the loader installs on the line immediately after, and the bridge
// publishes no window.__vrb.stop, so there is never a live handle to call. So
// duplicates accumulate, and every wake, every bail and every composer send
// happens once per instance.
//
// Each instance stamps its own badge into the page, so the duplicates are
// visible in the DOM: two or more badges means two or more instances. A reload
// tears all of them down at once and exactly one clean instance boots.
//
// The reload is throttled through sessionStorage, which survives a reload, so a
// page that keeps producing duplicates reloads at most once every twenty
// seconds instead of spinning.
//
// This does not prevent the second boot. It collapses the result, which is what
// stops the triple sends.

(function () {
'use strict';
var AT = 'vrb.guardAt', MIN = 20000;

function dup() {
var els = document.querySelectorAll('div'), w = 0, l = 0, i, t;
for (i = 0; i 40) continue;
if (t.indexOf('wakeg') === 0) w++;
else if (t.indexOf('LOAD') === 0) l++;
}
return w > 1 || l > 1;
}

function go() {
var now = Date.now(), prev = 0;
try { prev = Number(sessionStorage.getItem(AT) || 0); } catch (e) {}
if (now - prev < MIN) return;
try { sessionStorage.setItem(AT, String(now)); } catch (e) {}
console.log('[guard] duplicate instance, reloading');
location.reload();
}

setInterval(function () { try { if (dup()) go(); } catch (e) {} }, 5000);
})();
